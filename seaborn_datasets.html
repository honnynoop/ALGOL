Quick Start

import seaborn as sns

sns.get_dataset_names()           # 전체 목록 확인

df = sns.load_dataset('iris')      # 데이터셋 불러오기
df = sns.load_dataset('titanic')   # 분류 실습
df = sns.load_dataset('tips')      # 회귀 실습
df = sns.load_dataset('flights')   # 시계열 실습

# brain_networks 는 header=[0,1] 필요
df = sns.load_dataset('brain_networks', index_col=0, header=[0,1])
