검정 공통 해석 패턴
# 공통 패턴: 통계량, p-value 출력 후 판단
from scipy import stats
import numpy as np

# p-value 해석 함수
def interpret(stat, p, alpha=0.05):
    print(f"검정통계량: {stat:.4f}")
    print(f"p-value   : {p:.4f}")
    if p < alpha:
        print(f"→ p({p:.4f}) < α({alpha}) : H₀ 기각 ✓ (유의미한 차이)")
    else:
        print(f"→ p({p:.4f}) ≥ α({alpha}) : H₀ 채택 (유의미한 차이 없음)")
t검정 전체 흐름 (독립 2표본)
from scipy.stats import shapiro, levene, ttest_ind

# 1. 정규성 검정 (Shapiro-Wilk)
_, p_a = shapiro(group_a)
_, p_b = shapiro(group_b)
print(f"정규성 A: p={p_a:.4f}, B: p={p_b:.4f}")
# p > 0.05 → 정규분포 가정 성립
# 2. 등분산 검정 (Levene)
stat, p_var = levene(group_a, group_b)
print(f"등분산 검정: F={stat:.4f}, p={p_var:.4f}")
equal_var = p_var >= 0.05 # True → 등분산
# 3. 독립 2표본 t검정
t_stat, p_val = ttest_ind(group_a, group_b, equal_var=equal_var)
interpret(t_stat, p_val)

from scipy.stats import chi2_contingency
import pandas as pd

# 분할표 생성
table = pd.crosstab(df['A'], df['B'])

print(f"χ²={chi2:.4f}, df={dof}")

일원 ANOVA + 사후검정
from scipy.stats import f_oneway
from statsmodels.stats.multicomp import pairwise_tukeyhsd

F, p = f_oneway(g1, g2, g3)
interpret(F, p)

# ANOVA 유의 → 사후검정
result = pairwise_tukeyhsd(
    np.concatenate([g1,g2,g3]),
    np.repeat(['A','B','C'],[len(g1),len(g2),len(g3)])
)
print(result.summary())
비율 Z검정 (단일표본 / 2표본)
from statsmodels.stats.proportion import proportions_ztest, proportion_confint

# 단일표본 비율 검정: 성공 횟수, 전체 수, 귀무가설 비율
count, nobs, p0 = 45, 100, 0.5 # H₀: p = 0.5
z, p_val = proportions_ztest(count, nobs, p0)
interpret(z, p_val)

# 신뢰구간
ci_low, ci_upp = proportion_confint(count, nobs, alpha=0.05)
print(f"95% CI: [{ci_low:.4f}, {ci_upp:.4f}]")

# 2표본 비율 검정
z2, p2 = proportions_ztest([30, 40], [100, 100])
interpret(z2, p2)
상관분석 + 회귀분석
from scipy.stats import pearsonr, spearmanr
import statsmodels.api as sm

# 피어슨 상관
r, p = pearsonr(x, y)
print(f"피어슨 r={r:.4f}, p={p:.4f}")

# 스피어만 상관 (비모수)
rho, p_s = spearmanr(x, y)
print(f"스피어만 ρ={rho:.4f}, p={p_s:.4f}")

# 단순회귀
X = sm.add_constant(x)
model = sm.OLS(y, X).fit()
print(model.summary())
