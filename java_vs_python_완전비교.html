상속 (Inheritance)
☕ JAVA — extends
public class Student extends Person { private String major; public Student(String name, int age, String major) { super(name, age); // 부모 생성자 this.major = major; } // 메서드 오버라이드 @Override public void introduce() { super.introduce(); // 부모 메서드 System.out.println( "전공: " + major); } } // 추상 클래스 public abstract class Shape { abstract double area(); void print() { System.out.println("넓이:"+area()); } } // 인터페이스 interface Flyable { void fly(); default void land() { System.out.println("착지"); } }
🐍 PYTHON — 괄호 안에 부모 클래스

☕ JAVA — checked / unchecked
// 기본 try-catch-finally try { int result = 10 / 0; } catch (ArithmeticException e) { System.out.println( "산술 오류: " + e.getMessage()); } catch (Exception e) { System.out.println( "일반 오류: " + e.getMessage()); } finally { System.out.println("항상 실행"); } // 다중 예외 (Java 7+) try { // ... } catch (IOException | SQLException e) { e.printStackTrace(); } // 예외 발생 throw new IllegalArgumentException("잘못됨"); // 커스텀 예외 class MyException extends RuntimeException { public MyException(String msg) { super(msg); } } // try-with-resources (자동 close) try (FileReader fr = new FileReader("f.txt")) { // 자동으로 close() 호출 }
🐍 PYTHON — except 구문
# 기본 try-except-finally try: result = 10 / 0 except ZeroDivisionError as e: print(f"나누기 오류: {e}") except Exception as e: print(f"일반 오류: {e}") else: print("예외 없을 때 실행") # Java 없음 finally: print("항상 실행") # 다중 예외 except (ValueError, TypeError) as e: print(e) # 예외 발생 raise ValueError("잘못됨") # 커스텀 예외 class MyException(Exception): def __init__(self, msg): super().__init__(msg) # with문 (try-with-resources 상당) with open("f.txt") as f: data = f.read() # 자동으로 close()
✅
Python의 try-except-else: else 블록은 예외가 발생하지 않았을 때만 실행됩니다. Java에는 이 구문이 없습니다. try-except-else-finally 순서로 씁니다.

☕ JAVA
String s = " Hello, World! "; // 기본 조작 s.length() // 18 s.trim() // "Hello, World!" s.strip() // trim() + 유니코드 (Java 11+) s.toUpperCase() // " HELLO, WORLD! " s.toLowerCase() // 포함 / 검색 s.contains("World") // true s.startsWith(" H") // true s.endsWith(" ") // true s.indexOf("o") // 4 // 분리 / 결합 s.split(", ") // [" Hello","World! "] String.join("-", "a", "b", "c") // "a-b-c" // 치환 / 추출 s.replace("World", "Java") s.replaceAll("\\s+", " ") // 정규식 s.substring(2, 7) // "Hello" // 비교 s.equals("other") // == 아님! s.equalsIgnoreCase("hello") s.compareTo("other") // 사전순 비교 // 숫자 포맷 String fmt = String.format("%.2f", 3.14159);

실전 종합 예제
같은 프로그램을 Java와 Python으로 — 성적 처리 프로그램
☕ JAVA — 성적 처리
import java.util.*; public class GradeSystem { record Student(String name, int score) {} public static String getGrade(int score) { return switch (score / 10) { case 10, 9 -> "A"; case 8 -> "B"; case 7 -> "C"; case 6 -> "D"; default -> "F"; }; } public static void main(String[] args) { List<Student> students = List.of( new Student("Alice", 92), new Student("Bob", 75), new Student("Carol", 88), new Student("Dave", 54) ); // 평균 double avg = students.stream() .mapToInt(Student::score) .average().getAsDouble(); // 정렬 (점수 내림차순) students.stream() .sorted((a,b) -> b.score()-a.score()) .forEach(s -> System.out.printf( "%-8s %3d점 %s%n", s.name(), s.score(), getGrade(s.score()))); System.out.printf("평균: %.1f%n", avg); } }
🐍 PYTHON — 성적 처리
from dataclasses import dataclass @dataclass class Student: name: str score: int def get_grade(score: int) -> str: match score // 10: case 10 | 9: return "A" case 8: return "B" case 7: return "C" case 6: return "D" case _: return "F" students = [ Student("Alice", 92), Student("Bob", 75), Student("Carol", 88), Student("Dave", 54), ] # 평균 avg = sum(s.score for s in students) / len(students) # 정렬 (점수 내림차순) for s in sorted(students, key=lambda x: x.score, reverse=True): print(f"{s.name:8} {s.score:>3}점 {get_grade(s.score)}") print(f"평균: {avg:.1f}") if __name__ == "__main__": print("실행 완료")
