from sklearn.datasets import load_iris
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression
from sklearn.neighbors import KNeighborsClassifier
from sklearn.svm import SVC
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import cross_val_score
import pandas as pd

X, y = load_iris(return_X_y=True)

models = {
    'Logistic':      LogisticRegression(max_iter=300),
    'KNN(k=5)':      KNeighborsClassifier(n_neighbors=5),
    'SVM(RBF)':      SVC(kernel='rbf', C=1),
    'RandomForest':  RandomForestClassifier(n_estimators=100, random_state=42),
}

rows = []
for name, clf in models.items():
    pipe = Pipeline([('sc', StandardScaler()), ('clf', clf)])
    scores = cross_val_score(pipe, X, y, cv=5, scoring='accuracy')
    rows.append({'Model': name, 'Mean': scores.mean(), 'Std': scores.std()})

print(pd.DataFrame(rows).sort_values('Mean', ascending=False).round(4).to_string(index=False))
📊 결과 해석
  • SVM(RBF)·RF 모두 CV 정확도 ≈ 0.973 — iris는 쉬운 데이터셋
  • Pipeline 내부에서 각 fold마다 StandardScaler를 별도 fit → 데이터 누출 방지
  • LogisticRegression도 ≈ 0.960으로 선형 모델 대비 손색 없음
  • 다른 데이터셋으로 교체 시 Pipeline 구조 재사용 가능

from sklearn.datasets import load_iris
from sklearn.svm import SVC
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from sklearn.model_selection import GridSearchCV
import pandas as pd

X, y = load_iris(return_X_y=True)

pipe = Pipeline([('sc', StandardScaler()), ('svm', SVC())])
param_grid = {
    'svm__C':      [0.1, 1, 10, 100],
    'svm__kernel': ['rbf', 'linear'],
    'svm__gamma':  ['scale', 'auto', 0.01, 0.1],
}

gs = GridSearchCV(pipe, param_grid, cv=5,
                  scoring='accuracy', n_jobs=-1, return_train_score=True)
gs.fit(X, y)

print(f'최적 파라미터: {gs.best_params_}')
print(f'최적 CV 정확도: {gs.best_score_:.4f}')

# 상위 5개 결과
res = pd.DataFrame(gs.cv_results_)
cols = ['param_svm__C','param_svm__kernel','param_svm__gamma',
        'mean_test_score','std_test_score']
print(res.nlargest(5, 'mean_test_score')[cols].to_string(index=False))
📊 결과 해석
  • C=10, kernel='rbf', gamma='scale' 근방에서 최고 성능 달성
  • param_grid 키에 'svm__'처럼 Pipeline 스텝명을 접두어로 붙여야 함
  • return_train_score=True로 과적합 여부 확인 (train - test 점수 차이)
  • n_jobs=-1로 CPU 코어 전부 사용 — 파라미터 조합이 많을수록 속도 차이 큼
💡 팁: RandomizedSearchCV는 파라미터 공간이 클 때 GridSearch보다 효율적

from sklearn.datasets import load_iris
from sklearn.svm import SVC
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from sklearn.model_selection import learning_curve
import numpy as np

X, y = load_iris(return_X_y=True)
pipe = Pipeline([('sc', StandardScaler()), ('svm', SVC(kernel='rbf', C=1))])

train_sizes, train_scores, val_scores = learning_curve(
    pipe, X, y,
    train_sizes=np.linspace(0.1, 1.0, 10),
    cv=5, scoring='accuracy', n_jobs=-1)

print(f'{"Train Size":>10}  {"Train Acc":>10}  {"Val Acc":>10}  {"Gap":>8}')
for sz, tr, vl in zip(train_sizes,
                       train_scores.mean(1),
                       val_scores.mean(1)):
    print(f'{sz:>10d}  {tr:>10.4f}  {vl:>10.4f}  {tr-vl:>8.4f}')
📊 결과 해석
  • 학습 데이터가 늘어날수록 val_acc가 수렴 → 더 많은 데이터 수집이 도움됨
  • Gap(train-val)이 크면 과적합, 둘 다 낮으면 과소적합
  • iris는 150개로도 충분 — val_acc가 n≈50 이후 안정적으로 수렴
  • learning_curve는 '데이터를 더 모아야 하는가?'를 판단하는 핵심 도구
💡 팁: validation_curve는 학습 크기 대신 하나의 하이퍼파라미터를 변화시켜 최적값 탐색

PCA — 64D→2D 시각화
1표준화2PCA 2성분3분산 기여율 + 95% 유지 성분 수
Python 
from sklearn.datasets import load_digits
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
import numpy as np

X, y = load_digits(return_X_y=True)
X_sc = StandardScaler().fit_transform(X)

# 2D 시각화
pca2 = PCA(n_components=2, random_state=42)
X_2d = pca2.fit_transform(X_sc)
print(f'2성분 누적 기여율: {pca2.explained_variance_ratio_.sum():.4f}')

# 95% 분산 유지에 필요한 성분 수
pca_full = PCA(random_state=42).fit(X_sc)
cumvar = np.cumsum(pca_full.explained_variance_ratio_)
n95 = np.searchsorted(cumvar, 0.95) + 1
print(f'95% 분산 유지에 필요한 PCA 성분 수: {n95} / 64')
print(f'압축률: {n95/64*100:.1f}%')

# 첫 5개 성분 기여율
print('\n상위 5개 주성분 기여율:')
for i, v in enumerate(pca_full.explained_variance_ratio_[:5]):
    print(f'  PC{i+1}: {v:.4f} ({cumvar[i]:.4f} 누적)')
📊 결과 해석
  • 2D PCA 누적 기여율 ≈ 28% — 64차원을 2D로 내리면 정보 손실이 큼
  • 95% 분산 유지에 ≈ 41개 성분 필요 (64→41, 36% 압축)
  • 상위 PC1 혼자 약 12% 설명 — 픽셀 공간의 분산이 많은 차원에 분산되어 있음
  • 2D 산점도에서 같은 숫자끼리 군집을 이루나 일부 겹침 (0과 1은 잘 분리됨)
💡 팁: PCA 후 SVM/RF 분류 시 n95 성분을 사용하면 계산량 줄이면서 성능 유지 가능
SVM vs MLP — 고차원 분류 비교
1PCA 전처리 선택2SVM(RBF)·MLP 학습3분류 리포트 비교
Python 
from sklearn.datasets import load_digits
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
from sklearn.svm import SVC
from sklearn.neural_network import MLPClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, classification_report

X, y = load_digits(return_X_y=True)
X_tr, X_te, y_tr, y_te = train_test_split(
    X, y, test_size=0.2, stratify=y, random_state=42)

pipelines = {
    'SVM(RBF)': Pipeline([
        ('sc', StandardScaler()),
        ('svm', SVC(kernel='rbf', C=10, gamma='scale'))]),
    'MLP': Pipeline([
        ('sc', StandardScaler()),
        ('mlp', MLPClassifier(hidden_layer_sizes=(256, 128),
                               max_iter=500, random_state=42))]),
    'PCA(40)+SVM': Pipeline([
        ('sc', StandardScaler()),
        ('pca', PCA(n_components=40)),
        ('svm', SVC(kernel='rbf', C=10))]),
}

for name, pipe in pipelines.items():
    pipe.fit(X_tr, y_tr)
    acc = accuracy_score(y_te, pipe.predict(X_te))
    print(f'{name:15}: test acc = {acc:.4f}')
📊 결과 해석
  • SVM(RBF) 테스트 정확도 ≈ 0.989 — 64차원 픽셀에서도 매우 높은 성능
  • MLP(256,128) ≈ 0.978 — 더 많은 파라미터지만 SVM에 소폭 열세
  • PCA(40)+SVM도 ≈ 0.983 — 성능 거의 유지하면서 차원을 37% 축소
  • 이미지 분류에서 SVM+RBF는 픽셀 특성 공간에서 비선형 경계를 효과적으로 학습
💡 팁: MLPClassifier는 early_stopping=True, validation_fraction=0.1로 과적합 자동 방지 가능
혼동 행렬 + 오분류 분석
1예측 수행2혼동 행렬 계산3가장 많이 틀린 쌍 확인
Python 
from sklearn.datasets import load_digits
from sklearn.svm import SVC
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from sklearn.model_selection import train_test_split
from sklearn.metrics import confusion_matrix
import numpy as np
import pandas as pd

X, y = load_digits(return_X_y=True)
X_tr, X_te, y_tr, y_te = train_test_split(
    X, y, test_size=0.2, stratify=y, random_state=42)

pipe = Pipeline([('sc', StandardScaler()),
                 ('svm', SVC(kernel='rbf', C=10, gamma='scale'))])
pipe.fit(X_tr, y_tr)
y_pred = pipe.predict(X_te)

cm = confusion_matrix(y_te, y_pred)
# 대각(정답) 제외 후 가장 많이 혼동된 쌍
np.fill_diagonal(cm, 0)
errors = pd.DataFrame(
    [(i, j, cm[i,j]) for i in range(10) for j in range(10) if cm[i,j]>0],
    columns=['실제','예측','횟수']
).sort_values('횟수', ascending=False)
print('가장 많이 혼동된 쌍:')
print(errors.head(10).to_string(index=False))
📊 결과 해석
  • 9→4, 3→5, 8→1 등의 쌍이 가장 많이 혼동 — 형태가 비슷한 숫자
  • SVM 오분류는 전체 360개 테스트 중 보통 3~5개 수준 (정확도 ≈ 98.9%)
  • 혼동 행렬 대각 제외 후 정렬하면 모델이 어디서 실패하는지 직관적 파악

💡 팁: 오분류 이미지를 직접 확인하는 것이 모델 개선 방향을 찾는 가장 빠른 방법

from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
import numpy as np
from scipy.stats import shapiro, spearmanr

X, y = load_diabetes(return_X_y=True)
X_sc = StandardScaler().fit_transform(X)
X_tr, X_te, y_tr, y_te = train_test_split(X_sc, y, test_size=0.2, random_state=42)

residuals = y_te - y_pred

print(f'잔차 평균: {residuals.mean():.4f}  (0에 가까워야 함)')
print(f'잔차 표준편차: {residuals.std():.4f}')

print('정규 분포' if p > 0.05 else '비정규 분포 → 변환 고려')

rho, p_sp = spearmanr(y_pred, np.abs(residuals))

    preds = np.zeros_like(y_sc)
    for tr, te in loo.split(X_sc):

        pls.fit(X_sc[tr], y_sc[tr])
        preds[te] = pls.predict(X_sc[te])

    print(f'{n:>8}  {r2s[0]:>10.4f}  {r2s[1]:>10.4f}  {r2s[2]:>10.4f}')

pls2 = PLSRegression(n_components=2).fit(X_sc, y_sc)

from sklearn.datasets import make_circles
from sklearn.svm import SVC
from sklearn.neural_network import MLPClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from sklearn.model_selection import cross_val_score
import pandas as pd

rows = []
for factor in [0.3, 0.5, 0.7]:
    X, y = make_circles(n_samples=400, noise=0.1,
                         factor=factor, random_state=42)
    for name, clf in [
        ('Logistic(선형)',  LogisticRegression()),
        ('SVM(RBF)',       SVC(kernel='rbf', C=1)),
        ('MLP(50,30)',     MLPClassifier(hidden_layer_sizes=(50,30), max_iter=500)),
    ]:
        pipe = Pipeline([('sc', StandardScaler()), ('clf', clf)])
        acc = cross_val_score(pipe, X, y, cv=5, scoring='accuracy').mean()
        rows.append({'factor': factor, 'Model': name, 'Accuracy': acc})

df = pd.DataFrame(rows)
print(df.pivot_table('Accuracy', index='factor', columns='Model').round(4).to_string())

from sklearn.datasets import make_circles
from sklearn.linear_model import LogisticRegression
from sklearn.svm import SVC
from sklearn.model_selection import cross_val_score
import numpy as np

X, y = make_circles(n_samples=400, noise=0.05, factor=0.5, random_state=42)

# 반지름 피처 추가
r = np.sqrt((X ** 2).sum(axis=1, keepdims=True))
X_eng = np.hstack([X, r])

for name, X_use in [('원본 X(2D)', X), ('반지름 추가(3D)', X_eng)]:
    acc_lr  = cross_val_score(LogisticRegression(), X_use, y, cv=5).mean()
    acc_svm = cross_val_score(SVC(kernel='linear'), X_use, y, cv=5).mean()
    print(f'{name}')
    print(f'  Logistic: {acc_lr:.4f}')
    print(f'  SVM(linear): {acc_svm:.4f}')
📊 결과 해석
  • 원본 2D에서 선형 모델 정확도 ≈ 0.50 — 완전히 무작위 수준
  • r = √(x₁²+x₂²) 피처 추가 후 선형 Logistic 정확도 ≈ 0.95 이상
  • RBF 커널이 하는 일을 수동으로 피처 엔지니어링으로 모방 — '커널 트릭' 직관 이해
  • 선형 모델 + 좋은 피처 > 복잡한 비선형 모델 + 원본 피처 → 피처 엔지니어링의 중요성
💡 팁: 커널 트릭: 고차원 매핑을 명시적으로 계산하지 않고 내적만으로 표현 — 계산 효율

    'PCA':     PCA(n_components=2),
    'Isomap':  Isomap(n_neighbors=10, n_components=2),

from sklearn.datasets import make_swiss_roll
from sklearn.manifold import TSNE
import numpy as np
from scipy.stats import spearmanr

X, color = make_swiss_roll(n_samples=800, noise=0.1, random_state=42)

print(f'{"perplexity":>12}  {"n_iter":>8}  {"Spearman":>10}  {"KL_div":>10}')
for perp in [5, 15, 30, 50, 100]:
    tsne = TSNE(n_components=2, perplexity=perp,
                n_iter=1000, random_state=42)
    X_2d = tsne.fit_transform(X)
    rho, _ = spearmanr(color, X_2d[:, 0])
    print(f'{perp:>12}  {1000:>8}  {abs(rho):>10.4f}  {tsne.kl_divergence_:>10.4f}')
📊 결과 해석

  • perplexity가 너무 작으면 과도한 클러스터 분리, 너무 크면 구조가 뭉개짐
  • kl_divergence_가 낮을수록 원본 구조를 잘 보존하는 임베딩
  • n_iter=1000으로도 충분하나 복잡한 데이터는 3000~5000 필요
