Quick Reference 
from sklearn import datasets

# ── Toy (load_*) ──────────────────────────────
iris     = datasets.load_iris()            # 분류 기본
wine     = datasets.load_wine()            # 다중분류
cancer   = datasets.load_breast_cancer()  # 이진분류
diabetes = datasets.load_diabetes()        # 회귀 기본
digits   = datasets.load_digits()          # 이미지 분류

# ── Real-world (fetch_*) ──────────────────────
housing  = datasets.fetch_california_housing()
news     = datasets.fetch_20newsgroups(subset='train', categories=['sci.space'])
faces    = datasets.fetch_lfw_people(min_faces_per_person=70)
any_ds   = datasets.fetch_openml(name='titanic', version=1, as_frame=True)

# ── Generated (make_*) ────────────────────────
X, y = datasets.make_classification(n_samples=1000, n_features=20,
                                     n_classes=3, n_informative=5)
X, y = datasets.make_regression(n_samples=500, noise=0.1)
X, y = datasets.make_blobs(n_samples=300, centers=4, cluster_std=0.6)
X, y = datasets.make_moons(n_samples=200, noise=0.1)
X, y = datasets.make_circles(n_samples=200, noise=0.05, factor=0.5)
X, t = datasets.make_swiss_roll(n_samples=1000, noise=0.05)

# ── 공통 속성 ──────────────────────────────────
print(iris.data.shape)    # (150, 4)
print(iris.target_names)  # ['setosa', 'versicolor', 'virginica']
print(iris.feature_names) # 특성 이름
print(iris.DESCR)         # 상세 설명
