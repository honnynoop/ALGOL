# ── IQR 방법 (★ 가장 자주 출제) ────────────────────────── Q1 = df['col'].quantile(0.25) Q3 = df['col'].quantile(0.75) IQR = Q3 - Q1 lower = Q1 - 1.5 * IQR upper = Q3 + 1.5 * IQR # 이상치 제거 df_clean = df[(df['col'] >= lower) & (df['col'] <= upper)] # 이상치 대체 (Capping) df['col'] = df['col'].clip(lower=lower, upper=upper) # 이상치 개수 확인 outliers = df[(df['col'] < lower) | (df['col'] > upper)] print(len(outliers)) # ── Z-score 방법 ───────────────────────────────────────── from scipy.stats import zscore z = np.abs(zscore(df['col'])) df_clean = df[z < 3] # |z| < 3 이면 정상 # 여러 컬럼에 적용 z_scores = df.apply(zscore) df_clean = df[(z_scores.abs() < 3).all(axis=1)]
