3 PCA 주성분 분석 4차원 → 2차원 시각화
PCA란?
Principal Component Analysis(주성분 분석). 고차원 데이터의 분산을 최대한 보존하는 새 축(주성분)을 찾아 저차원으로 압축합니다. 4개 특성을 2개 PC로 줄여도 전체 분산의 97.8%를 유지합니다.
PCA 2D 산점도 (PC1 vs PC2)
Setosa
Versicolor
Virginica

PC1축만으로 Setosa 완전 분리 / PC2가 Versicolor·Virginica 구분 보조

주성분별 설명 분산 비율 (Explained Variance Ratio)

PC1: 72.96% + PC2: 22.85% = 누적 95.81% — 2개 PC로 거의 완벽 표현

PCA 로딩 행렬 — 각 주성분에 대한 원본 특성 기여도

PC1: petal 두 특성이 지배 (분류력 높음) / PC2: sepal width가 반대 방향으로 기여

PCA 분석 코드 Step 3 
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt

X = iris.data
y = iris.target

# 1. 표준화 (PCA 전 필수 — 단위/스케일 차이 제거)
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 2. PCA 학습
pca = PCA(n_components=4)   # 전체 주성분 확인
X_pca = pca.fit_transform(X_scaled)

# 3. 설명 분산 비율
print("설명 분산 비율:", pca.explained_variance_ratio_.round(4))
# [0.7296, 0.2285, 0.0367, 0.0052]
print("누적 분산 비율:", np.cumsum(pca.explained_variance_ratio_).round(4))
# [0.7296, 0.9581, 0.9948, 1.0000]

# 4. 로딩 행렬 (특성별 기여도)
loading_df = pd.DataFrame(pca.components_.T,
    index=iris.feature_names,
    columns=[f'PC{i+1}' for i in range(4)])
print(loading_df.round(3))

# 5. 2D 시각화
pca2 = PCA(n_components=2)
X_2d = pca2.fit_transform(X_scaled)
colors = ['#5DCAA5', '#7F77DD', '#D85A30']
fig, ax = plt.subplots(figsize=(8, 6))
for i, name in enumerate(iris.target_names):
    mask = y == i
    ax.scatter(X_2d[mask, 0], X_2d[mask, 1],
               c=colors[i], label=name, alpha=0.75, s=50, edgecolors='white', linewidths=0.5)
ax.set_xlabel(f"PC1 ({pca2.explained_variance_ratio_[0]*100:.1f}%)")
ax.set_ylabel(f"PC2 ({pca2.explained_variance_ratio_[1]*100:.1f}%)")
ax.legend()
ax.set_title('Iris PCA 2D 시각화')
plt.tight_layout()
plt.show()

분류 모델 전체 코드 Step 4 
from sklearn.model_selection import train_test_split, cross_val_score, StratifiedKFold
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression
from sklearn.svm import SVC
from sklearn.tree import DecisionTreeClassifier
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import classification_report, confusion_matrix
import numpy as np

X, y = iris.data, iris.target

# 1. 분할 + 표준화
X_tr, X_te, y_tr, y_te = train_test_split(
    X, y, test_size=0.3, random_state=42, stratify=y)

scaler = StandardScaler()
X_tr_s = scaler.fit_transform(X_tr)
X_te_s = scaler.transform(X_te)     # 주의: transform만! fit_transform 금지

# 2. 모델 정의
models = {
    'Logistic Regression': LogisticRegression(max_iter=200),
    'SVM (RBF)':           SVC(kernel='rbf', C=1.0),
    'K-NN (k=5)':          KNeighborsClassifier(n_neighbors=5),
    'Decision Tree':       DecisionTreeClassifier(random_state=42),
    'Random Forest':       RandomForestClassifier(n_estimators=100, random_state=42),
}

# 3. 교차검증 비교
cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
for name, model in models.items():
    scores = cross_val_score(model, X_tr_s, y_tr, cv=cv, scoring='accuracy')
    print(f"{name:25s}: {scores.mean():.4f} ± {scores.std():.4f}")

# 4. 최고 모델(SVM) 상세 평가
best = SVC(kernel='rbf', C=1.0)
best.fit(X_tr_s, y_tr)
y_pred = best.predict(X_te_s)

print(classification_report(y_te, y_pred, target_names=iris.target_names))
print(confusion_matrix(y_te, y_pred))
PCA + 분류 파이프라인 (Pipeline) 심화 
from sklearn.pipeline import Pipeline
from sklearn.decomposition import PCA
from sklearn.svm import SVC

# StandardScaler → PCA(2) → SVM 파이프라인
pipe = Pipeline([
    ('scaler', StandardScaler()),
    ('pca',    PCA(n_components=2)),
    ('clf',    SVC(kernel='rbf', C=1.0))
])

pipe.fit(X_tr, y_tr)
print("PCA(2) + SVM 정확도:", pipe.score(X_te, y_te))
# 약 0.956 — 4특성 SVM과 거의 동일!

# GridSearchCV로 최적 n_components 탐색
from sklearn.model_selection import GridSearchCV

param_grid = {
    'pca__n_components': [1, 2, 3, 4],
    'clf__C':            [0.1, 1, 10],
    'clf__kernel':       ['linear', 'rbf']
}
gs = GridSearchCV(pipe, param_grid, cv=5, scoring='accuracy')
gs.fit(X_tr, y_tr)
print("최적 파라미터:", gs.best_params_)
print("최고 CV 점수:", gs.best_score_.round(4))

# ─────────────────────────────────────────────────────
# Iris 완전 분석: EDA → PCA → 다중분류 → 평가
# ─────────────────────────────────────────────────────
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.datasets import load_iris
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.decomposition import PCA
from sklearn.model_selection import train_test_split, cross_val_score, StratifiedKFold
from sklearn.linear_model import LogisticRegression
from sklearn.svm import SVC
from sklearn.tree import DecisionTreeClassifier
from sklearn.ensemble import RandomForestClassifier
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import classification_report, confusion_matrix, ConfusionMatrixDisplay
from sklearn.pipeline import Pipeline

# ── 0. 데이터 로드 ────────────────────────────────────
iris = load_iris()
X, y = iris.data, iris.target
df = pd.DataFrame(X, columns=iris.feature_names)
df['species'] = pd.Categorical.from_codes(y, iris.target_names)
print("shape:", df.shape)
print(df.describe().round(2))

# ── 1. EDA ────────────────────────────────────────────
fig, axes = plt.subplots(1, 4, figsize=(16, 4))
for i, feat in enumerate(iris.feature_names):
    df.boxplot(column=feat, by='species', ax=axes[i])
    axes[i].set_title(feat, fontsize=9)
plt.suptitle(''); plt.tight_layout(); plt.show()

# ── 2. 표준화 + PCA ───────────────────────────────────
scaler = StandardScaler()
X_s = scaler.fit_transform(X)

pca4 = PCA(n_components=4)
pca4.fit(X_s)
evr = pca4.explained_variance_ratio_
print("설명 분산:", np.round(evr, 4))
print("누적 분산:", np.round(np.cumsum(evr), 4))

pca2 = PCA(n_components=2)
X_2d = pca2.fit_transform(X_s)

plt.figure(figsize=(8, 6))
colors = ['#5DCAA5', '#7F77DD', '#D85A30']
for i, name in enumerate(iris.target_names):
    m = y == i
    plt.scatter(X_2d[m,0], X_2d[m,1], c=colors[i], label=name,
                alpha=0.75, s=55, edgecolors='white', linewidths=0.5)
plt.xlabel(f"PC1 ({evr[0]*100:.1f}%)")
plt.ylabel(f"PC2 ({evr[1]*100:.1f}%)")
plt.title('PCA 2D'); plt.legend(); plt.tight_layout(); plt.show()

# ── 3. 분류 모델 비교 ─────────────────────────────────
X_tr, X_te, y_tr, y_te = train_test_split(
    X, y, test_size=0.3, random_state=42, stratify=y)
X_tr_s = scaler.fit_transform(X_tr)
X_te_s  = scaler.transform(X_te)

models = {
    'Logistic Regression': LogisticRegression(max_iter=200),
    'SVM (RBF)':          SVC(kernel='rbf'),
    'K-NN (k=5)':         KNeighborsClassifier(5),
    'Decision Tree':      DecisionTreeClassifier(random_state=42),
    'Random Forest':      RandomForestClassifier(100, random_state=42),
}

cv = StratifiedKFold(5, shuffle=True, random_state=42)
results = {}
for name, m in models.items():
    sc = cross_val_score(m, X_tr_s, y_tr, cv=cv)
    results[name] = sc
    print(f"{name:22}: {sc.mean():.4f} ± {sc.std():.4f}")

# ── 4. 최고 모델 상세 평가 (SVM) ─────────────────────
svm = SVC(kernel='rbf', C=1.0)
svm.fit(X_tr_s, y_tr)
y_pred = svm.predict(X_te_s)

print(classification_report(y_te, y_pred, target_names=iris.target_names))
cm = confusion_matrix(y_te, y_pred)
disp = ConfusionMatrixDisplay(cm, display_labels=iris.target_names)
disp.plot(cmap='Blues')
plt.title('Confusion Matrix — SVM')
plt.tight_layout()
plt.show()

시험 활용 포인트 (빅분기·ADP)
1. 전처리: StandardScaler → PCA 순서 (스케일링 없이 PCA 하면 분산 큰 특성이 PC1을 독점)
2. train/test 분리 후 scaler.fit_transform(X_tr) + scaler.transform(X_te) — X_te에 fit 금지 (데이터 누수)
3. 교차검증: stratify=y로 클래스 비율 유지, StratifiedKFold 사용
4. 평가지표: 균등 클래스이므로 Accuracy로 충분 — 불균형이면 F1 사용
5. Pipeline: 전처리+모델을 묶으면 CV에서 데이터 누수 자동 방지
